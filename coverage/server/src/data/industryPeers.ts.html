/** 去掉行业名中的罗马数字序号（Ⅰ/Ⅱ/Ⅲ/Ⅳ/V…），便于关键词匹配 */
function normalizeIndustryName(name: string): string {
  return name.replace(/[Ⅰ-Ⅹ]/g, '').replace(/\s+/g, '').trim();
}
