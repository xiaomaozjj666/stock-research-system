/** 归一化股票名：去前缀 + 仅保留中文字符，便于子串/模糊匹配 */
export function normalizeName(name: string): string {
  let n = name.trim();
  for (const p of NAME_PREFIXES) {
    if (n.startsWith(p)) {
      n = n.slice(p.length);
      break;
    }
  }
  return n.replace(/[^一-龥]/g, '');
}
